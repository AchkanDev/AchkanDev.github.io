class AuthenticateUserUseCase implements UseCase<AuthSession, AuthCredentials> {
  final AuthRepository _repository;
  final TokenStorageService _storage;

  const AuthenticateUserUseCase(this._repository, this._storage);

  @override
  Future<Result<AuthSession>> call(AuthCredentials credentials) async {
    final result = await _repository.authenticate(credentials);
    return result.when(
      success: (session) async {
        await _storage.persistTokens(session.tokens);
        return Result.ok(session);
      },
      failure: (failure) => Result.error(failure),
    );
  }
}

#version 460 core
#include <flutter/runtime_effect.glsl>

uniform vec2 uResolution;
uniform float uTime;
uniform sampler2D uTexture;
out vec4 fragColor;

void main() {
  vec2 uv = FlutterFragCoord().xy / uResolution;
  vec2 distortion = vec2(
    sin(uv.y * 12.0 + uTime * 1.8) * 0.012,
    cos(uv.x * 12.0 + uTime * 1.8) * 0.012
  );
  fragColor = texture(uTexture, uv + distortion);
}

class OfflineSyncCoordinator {
  final Isar _isar;
  final ApiClient _remoteClient;

  Future<void> reconcilePendingMutations() async {
    final pendingQueue = await _isar.syncEntities
      .filter().statusEqualTo(SyncStatus.pending).findAll();

    if (pendingQueue.isEmpty) return;

    // Offload deserialization & encryption to background Isolate
    await Isolate.run(() => SyncWorker.batchProcess(pendingQueue));
  }
}
